8. Time Between Impressions

time_between_impressions <- videos %>% 
    filter(!is.na(first_impression_at)) %>% 
    filter(!is.na(last_impression_at)) %>% 
    mutate(time_between_first_and_last_impression = last_impression_at - first_impression_at) %>% 
    mutate(hours_between_first_and_last_impression = time_between_first_and_last_impression / 3600) %>%
    mutate(days_between_first_and_last_impression = hours_between_first_and_last_impression / 24) %>% 
    select(creator_id, video_id, impressions_n, days_between_first_and_last_impression, hours_between_first_and_last_impression) %>%
    arrange(desc(hours_between_first_and_last_impression))
